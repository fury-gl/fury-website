    all_lines = actor.streamlines(
        lines,
        colors=(0.7, 0.7, 0.7),
        thickness=2.0,
        opacity=0.2,
    )

    roi_filtered = actor.streamlines(
        lines,
        colors=(0.0, 0.0, 1.0),
        thickness=4.0,
        outline_thickness=1.5,
        outline_color=(0.1, 0.1, 0.1),
        roi_mask=mask,
        roi_origin=roi_origin,
        opacity=1.0,
    )

    roi_surface = actor.contour_from_roi(
        mask,
        color=(0.0, 0.8, 0.3),
        opacity=0.4,
    )
    roi_surface.translate(roi_origin)
